```python
# logic.py
class Counter:
    def __init__(self):
        self.value = 0
    def increment(self):
        self.value += 1
        return self.value

# test_logic.py
import unittest
class TestCounter(unittest.TestCase):
    def test_three_clicks(self):
        c = Counter()
        for _ in range(3):
            c.increment()
        self.assertEqual(c.value, 3)

# gui_app.py (اجرا در محیط گرافیکی)
import tkinter as tk
counter = Counter()
root = tk.Tk()
label = tk.Label(root, text="۰")
label.pack()
def on_click():
    label.config(text=str(counter.increment()))
tk.Button(root, text="بشمار", command=on_click).pack()
root.mainloop()
```
